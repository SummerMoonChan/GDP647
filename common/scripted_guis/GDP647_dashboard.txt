        context_type = player_context
        parent_window_token = top_bar
        window_name = "GDP647_launcher"
        dirty = global.GDP647_ui_revision
        ai_enabled = { always = no }
        visible = { exists = yes }
        effects = {
            GDP647_toggle_click = {
                if = { limit = { has_country_flag = GDP647_open } GDP647_close_dashboard = yes }
                else = {
                    if = {
                        limit = { NOT = { has_global_flag = GDP647_ready_v2 } }
                        GDP647_initialize = yes
                    }
                    GDP647_repair_population_update_snapshot = yes
                    GDP647_backfill_population_display = yes
                    if = {
                        limit = { NOT = { has_global_flag = GDP647_world_totals_ready } }
                        GDP647_publish_economic_benchmarks = yes
                    }
                    if = {
                        limit = { NOT = { has_global_flag = GDP647_compact_amounts_ready_v2 } }
                        for_each_scope_loop = {
                            array = global.GDP647_countries
                            GDP647_format_country_amounts = yes
                        }
                        every_state = {
                            limit = { has_variable = GDP647_state_gdp }
                            GDP647_format_state_amounts = yes
                        }
                        set_temp_variable = { GDP647_record_faction_snapshot = 0 }
                        GDP647_refresh_faction_directory = yes
                        set_global_flag = GDP647_compact_amounts_ready_v2
                    }
                    GDP647_initialize_follow_cache = yes
                    if = {
                        limit = { NOT = { has_global_flag = GDP647_faction_card_ranks_ready } }
                        GDP647_rank_follow_factions = yes
                    }
                    set_country_flag = GDP647_open

                    GDP647_refresh_open_view = yes
                }
                add_to_variable = { global.GDP647_ui_revision = 1 }
            }
        }
    }
