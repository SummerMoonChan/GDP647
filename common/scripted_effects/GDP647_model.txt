# Freeze inputs before computing the four batches of states.
GDP647_geo_P5_capture_state = {
    set_variable = { GDP647_capture_controller = PREV.id }
    set_variable = { GDP647_capture_population_k = state_population_k }
    set_variable = { GDP647_geo_U9_f_pop = GDP647_capture_population_k }
    divide_variable = { GDP647_geo_U9_f_pop = 1000 }
    clamp_variable = { var = GDP647_geo_U9_f_pop min = 0 }
    set_variable = { GDP647_geo_P5_is_core = 0 }
    set_variable = { GDP647_geo_operation = 1 }
    set_variable = { GDP647_geo_access = 1 }
    if = {
        limit = { is_core_of = PREV }
        set_variable = { GDP647_geo_P5_is_core = 1 }
    }
    else = {
        set_variable = { GDP647_geo_operation = compliance }
        clamp_variable = { var = GDP647_geo_operation min = 0 max = 100 }
        multiply_variable = { GDP647_geo_operation = global.GDP647_p_occupation_step }
        add_to_variable = { GDP647_geo_operation = global.GDP647_p_occupation_base }
        set_variable = { GDP647_geo_access = GDP647_geo_operation }
        if = {
            limit = { NOT = { is_owned_by = PREV } }
            set_variable = { GDP647_geo_access = modifier@local_factories }
            add_to_variable = { GDP647_geo_access = 1 }
            clamp_variable = { var = GDP647_geo_access min = 0 max = 1 }
        }
    }
    set_variable = { GDP647_geo_civ_count = non_damaged_building_level@industrial_complex }
    clamp_variable = { var = GDP647_geo_civ_count min = 0 }
    set_variable = { GDP647_geo_mil_count = non_damaged_building_level@arms_factory }
    clamp_variable = { var = GDP647_geo_mil_count min = 0 }
    set_variable = { GDP647_geo_dock_count = non_damaged_building_level@dockyard }
    clamp_variable = { var = GDP647_geo_dock_count min = 0 }
    set_variable = { GDP647_geo_refinery_count = non_damaged_building_level@synthetic_refinery }
    clamp_variable = { var = GDP647_geo_refinery_count min = 0 }
    set_variable = { GDP647_geo_infra_count = non_damaged_building_level@infrastructure }
    clamp_variable = { var = GDP647_geo_infra_count min = 0 }
    set_variable = { GDP647_geo_infra_square = GDP647_geo_infra_count }
    add_to_variable = { GDP647_geo_infra_square = 1 }
    divide_variable = { GDP647_geo_infra_square = 4 }
    multiply_variable = { GDP647_geo_infra_square = GDP647_geo_infra_square }

    # x = CIV / (CIV + population), with a population floor of 250,000.
    set_temp_variable = { GDP647_root_input = GDP647_geo_U9_f_pop }
    clamp_temp_variable = { var = GDP647_root_input min = 0.25 }
    add_to_temp_variable = { GDP647_root_input = GDP647_geo_civ_count }
    set_temp_variable = { GDP647_root_denominator = GDP647_root_input }
    set_temp_variable = { GDP647_root_input = GDP647_geo_civ_count }
    divide_temp_variable = { GDP647_root_input = GDP647_root_denominator }
    GDP647_sqrt_0_to_1 = yes
    set_variable = { GDP647_geo_local_support = GDP647_root_result }

    set_variable = { GDP647_geo_resource_value = 0 }
    set_temp_variable = { GDP647_resource_part = resource@coal }
    clamp_temp_variable = { var = GDP647_resource_part min = 0 }
    multiply_temp_variable = { GDP647_resource_part = global.GDP647_p_resource_coal }
    add_to_variable = { GDP647_geo_resource_value = GDP647_resource_part }
    set_temp_variable = { GDP647_resource_part = resource@oil }
    clamp_temp_variable = { var = GDP647_resource_part min = 0 }
    multiply_temp_variable = { GDP647_resource_part = global.GDP647_p_resource_oil }
    add_to_variable = { GDP647_geo_resource_value = GDP647_resource_part }
    set_temp_variable = { GDP647_resource_part = resource@steel }
    clamp_temp_variable = { var = GDP647_resource_part min = 0 }
    multiply_temp_variable = { GDP647_resource_part = global.GDP647_p_resource_steel }
    add_to_variable = { GDP647_geo_resource_value = GDP647_resource_part }
    set_temp_variable = { GDP647_resource_part = resource@aluminium }
    clamp_temp_variable = { var = GDP647_resource_part min = 0 }
    multiply_temp_variable = { GDP647_resource_part = global.GDP647_p_resource_aluminium }
    add_to_variable = { GDP647_geo_resource_value = GDP647_resource_part }
    set_temp_variable = { GDP647_resource_part = resource@rubber }
    clamp_temp_variable = { var = GDP647_resource_part min = 0 }
    multiply_temp_variable = { GDP647_resource_part = global.GDP647_p_resource_rubber }
    add_to_variable = { GDP647_geo_resource_value = GDP647_resource_part }
    set_temp_variable = { GDP647_resource_part = resource@tungsten }
    clamp_temp_variable = { var = GDP647_resource_part min = 0 }
    multiply_temp_variable = { GDP647_resource_part = global.GDP647_p_resource_tungsten }
    add_to_variable = { GDP647_geo_resource_value = GDP647_resource_part }
    set_temp_variable = { GDP647_resource_part = resource@chromium }
    clamp_temp_variable = { var = GDP647_resource_part min = 0 }
    multiply_temp_variable = { GDP647_resource_part = global.GDP647_p_resource_chromium }
    add_to_variable = { GDP647_geo_resource_value = GDP647_resource_part }

    # Temp variables persist through the state loop, so use distinct names from the country totals.
    set_temp_variable = { GDP647_state_population_contribution = GDP647_geo_U9_f_pop }
    multiply_temp_variable = { GDP647_state_population_contribution = GDP647_geo_operation }
    add_to_variable = { PREV.GDP647_effective_population = GDP647_state_population_contribution }
    set_temp_variable = { GDP647_state_civ_contribution = GDP647_geo_civ_count }
    multiply_temp_variable = { GDP647_state_civ_contribution = GDP647_geo_access }
    add_to_variable = { PREV.GDP647_effective_civ = GDP647_state_civ_contribution }
    set_temp_variable = { GDP647_core_basis = GDP647_geo_U9_f_pop }
    if = {
        limit = { check_variable = { GDP647_geo_P5_is_core = 1 } }
        add_to_variable = { PREV.GDP647_next_core_population = GDP647_core_basis }
        set_temp_variable = { GDP647_core_infra = GDP647_geo_infra_count }
        multiply_temp_variable = { GDP647_core_infra = GDP647_core_basis }
        add_to_variable = { PREV.GDP647_core_infra_population = GDP647_core_infra }
        multiply_temp_variable = { GDP647_core_basis = GDP647_geo_urban }
        add_to_variable = { PREV.GDP647_core_urban_population = GDP647_core_basis }
    }

    add_to_array = { array = PREV.GDP647_geo_P5_states }
    if = {
        limit = { check_variable = { global.GDP647_geo_P5_bucket = 0 } }
        add_to_array = { array = PREV.GDP647_geo_P5_week_1 }
    }
    else_if = {
        limit = { check_variable = { global.GDP647_geo_P5_bucket = 1 } }
        add_to_array = { array = PREV.GDP647_geo_P5_week_2 }
    }
    else_if = {
        limit = { check_variable = { global.GDP647_geo_P5_bucket = 2 } }
        add_to_array = { array = PREV.GDP647_geo_P5_week_3 }
    }
    else = { add_to_array = { array = PREV.GDP647_geo_P5_week_4 } }
    add_to_variable = { global.GDP647_geo_P5_bucket = 1 }
    if = {
        limit = { check_variable = { global.GDP647_geo_P5_bucket = 4 } }
        set_variable = { global.GDP647_geo_P5_bucket = 0 }
    }
}

# Newton's method. Nonzero inputs here have at least one factory.
GDP647_sqrt_0_to_1 = {
    set_temp_variable = { GDP647_root_result = 0 }
    if = {
        limit = { check_variable = { GDP647_root_input > 0 } }
        set_temp_variable = { GDP647_root_result = 1 }
        set_temp_variable = { GDP647_root_steps = 0 }
        while_loop_effect = {
            limit = { check_variable = { GDP647_root_steps < 8 } }
            set_temp_variable = { GDP647_root_step = GDP647_root_input }
            divide_temp_variable = { GDP647_root_step = GDP647_root_result }
            add_to_temp_variable = { GDP647_root_result = GDP647_root_step }
            multiply_temp_variable = { GDP647_root_result = 0.5 }
            add_to_temp_variable = { GDP647_root_steps = 1 }
        }
    }
}

# The non-core reference passes the same observed resources and full operating access.
GDP647_calculate_state_sectors = {
    set_temp_variable = { GDP647_sector_civ_count = GDP647_geo_civ_count }
    multiply_temp_variable = { GDP647_sector_civ_count = GDP647_sector_access }
    set_temp_variable = { GDP647_sector_dock_value = GDP647_geo_dock_count }
    multiply_temp_variable = { GDP647_sector_dock_value = GDP647_sector_access }
    multiply_temp_variable = { GDP647_sector_dock_value = PREV.GDP647_dock_productivity }
    multiply_temp_variable = { GDP647_sector_dock_value = global.GDP647_p_dock_value }

    set_temp_variable = { GDP647_rural_share = 1 }
    subtract_from_temp_variable = { GDP647_rural_share = GDP647_geo_urban }
    set_variable = { GDP647_geo_agri = GDP647_geo_U9_f_pop }
    multiply_variable = { GDP647_geo_agri = GDP647_rural_share }
    multiply_variable = { GDP647_geo_agri = GDP647_sector_operation }
    set_temp_variable = { GDP647_farm_industry = GDP647_sector_civ_count }
    multiply_temp_variable = { GDP647_farm_industry = global.GDP647_p_farm_share }
    multiply_temp_variable = { GDP647_farm_industry = global.GDP647_p_farm_civ_weight }
    add_to_variable = { GDP647_geo_agri = GDP647_farm_industry }
    multiply_variable = { GDP647_geo_agri = GDP647_geo_infra_square }
    multiply_variable = { GDP647_geo_agri = global.GDP647_p_R1 }

    set_temp_variable = { GDP647_civil_remaining = 1 }
    subtract_from_temp_variable = { GDP647_civil_remaining = global.GDP647_p_farm_share }
    set_temp_variable = { GDP647_civil_urban = GDP647_geo_urban }
    multiply_temp_variable = { GDP647_civil_urban = 0.5 }
    add_to_temp_variable = { GDP647_civil_urban = 0.5 }
    set_temp_variable = { GDP647_civil_infra = GDP647_geo_local_support }
    add_to_temp_variable = { GDP647_civil_infra = 0.35 }
    multiply_temp_variable = { GDP647_civil_infra = GDP647_geo_infra_square }
    clamp_temp_variable = { var = GDP647_civil_infra max = 1.85 }
    set_temp_variable = { GDP647_civil_dock = GDP647_sector_dock_value }
    multiply_temp_variable = { GDP647_civil_dock = global.GDP647_p_dock_civ_share }
    set_variable = { GDP647_geo_civ = GDP647_sector_civ_count }
    multiply_variable = { GDP647_geo_civ = GDP647_civil_remaining }
    multiply_variable = { GDP647_geo_civ = GDP647_civil_urban }
    multiply_variable = { GDP647_geo_civ = PREV.GDP647_civil_productivity }
    multiply_variable = { GDP647_geo_civ = GDP647_civil_infra }
    add_to_variable = { GDP647_geo_civ = GDP647_civil_dock }
    multiply_variable = { GDP647_geo_civ = global.GDP647_p_R2 }

    set_temp_variable = { GDP647_military_dock = GDP647_sector_dock_value }
    multiply_temp_variable = { GDP647_military_dock = global.GDP647_p_dock_mil_share }
    set_variable = { GDP647_geo_mil = GDP647_geo_mil_count }
    multiply_variable = { GDP647_geo_mil = GDP647_sector_access }
    multiply_variable = { GDP647_geo_mil = PREV.GDP647_military_productivity }
    add_to_variable = { GDP647_geo_mil = GDP647_military_dock }
    multiply_variable = { GDP647_geo_mil = global.GDP647_p_R3 }

    set_temp_variable = { GDP647_service_support = GDP647_geo_local_support }
    multiply_temp_variable = { GDP647_service_support = 0.35 }
    add_to_temp_variable = { GDP647_service_support = PREV.GDP647_national_service_support }
    subtract_from_temp_variable = { GDP647_service_support = 1 }
    multiply_temp_variable = { GDP647_service_support = global.GDP647_p_service_industry_weight }
    add_to_temp_variable = { GDP647_service_support = 1 }
    set_variable = { GDP647_geo_service = GDP647_geo_U9_f_pop }
    multiply_variable = { GDP647_geo_service = GDP647_sector_operation }
    multiply_variable = { GDP647_geo_service = GDP647_geo_urban }
    multiply_variable = { GDP647_geo_service = GDP647_geo_urban }
    divide_variable = { GDP647_geo_service = 0.6 }
    multiply_variable = { GDP647_geo_service = PREV.GDP647_service_factor }
    multiply_variable = { GDP647_geo_service = GDP647_service_support }
    multiply_variable = { GDP647_geo_service = global.GDP647_p_R4 }

    set_variable = { GDP647_geo_resources = GDP647_geo_refinery_count }
    multiply_variable = { GDP647_geo_resources = 48 }
    multiply_variable = { GDP647_geo_resources = GDP647_sector_access }
    multiply_variable = { GDP647_geo_resources = PREV.GDP647_refinery_productivity }
    multiply_variable = { GDP647_geo_resources = global.GDP647_p_fuel_weight }
    add_to_variable = { GDP647_geo_resources = GDP647_geo_resource_value }
    multiply_variable = { GDP647_geo_resources = global.GDP647_p_R5 }

    set_variable = { GDP647_geo_P5_next_state_total = GDP647_geo_agri }
    add_to_variable = { GDP647_geo_P5_next_state_total = GDP647_geo_civ }
    add_to_variable = { GDP647_geo_P5_next_state_total = GDP647_geo_mil }
    add_to_variable = { GDP647_geo_P5_next_state_total = GDP647_geo_service }
    add_to_variable = { GDP647_geo_P5_next_state_total = GDP647_geo_resources }
}
