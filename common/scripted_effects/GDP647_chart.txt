GDP647_scale_chart_samples = {
    set_temp_variable = { GDP647_chart_low = GDP647_chart_display_samples^0 }
    set_temp_variable = { GDP647_chart_high = GDP647_chart_low }
    set_temp_variable = { GDP647_chart_index = 1 }
    while_loop_effect = {
        limit = { check_variable = { GDP647_chart_index < GDP647_chart_count } }
        set_temp_variable = { GDP647_chart_value = GDP647_chart_display_samples^GDP647_chart_index }
        clamp_temp_variable = { var = GDP647_chart_low max = GDP647_chart_value }
        clamp_temp_variable = { var = GDP647_chart_high min = GDP647_chart_value }
        add_to_temp_variable = { GDP647_chart_index = 1 }
    }
    if = {
        limit = { check_variable = { GDP647_chart_high = GDP647_chart_low } }
        set_temp_variable = { GDP647_chart_padding = GDP647_chart_high }
        multiply_temp_variable = { GDP647_chart_padding = 0.05 }
        clamp_temp_variable = { var = GDP647_chart_padding min = 0.001 }
        subtract_from_temp_variable = { GDP647_chart_low = GDP647_chart_padding }
        clamp_temp_variable = { var = GDP647_chart_low min = 0 }
        add_to_temp_variable = { GDP647_chart_high = GDP647_chart_padding }
    }
    set_temp_variable = { GDP647_chart_range = GDP647_chart_high }
    subtract_from_temp_variable = { GDP647_chart_range = GDP647_chart_low }
    set_temp_variable = { GDP647_chart_axis = GDP647_chart_high }
    GDP647_format_chart_axis = yes
    set_variable = { GDP647_chart_max_compact = GDP647_chart_axis }
    set_variable = { GDP647_chart_max_unit = GDP647_chart_axis_unit }
    set_temp_variable = { GDP647_chart_axis = GDP647_chart_low }
    GDP647_format_chart_axis = yes
    set_variable = { GDP647_chart_min_compact = GDP647_chart_axis }
    set_variable = { GDP647_chart_min_unit = GDP647_chart_axis_unit }
    set_temp_variable = { GDP647_chart_index = 0 }
    while_loop_effect = {
        limit = { check_variable = { GDP647_chart_index < GDP647_chart_count } }
        set_temp_variable = { GDP647_chart_value = GDP647_chart_display_samples^GDP647_chart_index }
        set_temp_variable = { GDP647_chart_y = GDP647_chart_high }
        subtract_from_temp_variable = { GDP647_chart_y = GDP647_chart_value }
        divide_temp_variable = { GDP647_chart_y = GDP647_chart_range }
        multiply_temp_variable = { GDP647_chart_y = 40 }
        round_temp_variable = GDP647_chart_y
        clamp_temp_variable = { var = GDP647_chart_y min = 0 max = 40 }
        add_to_array = { array = GDP647_chart_plot_points value = GDP647_chart_y }
        add_to_temp_variable = { GDP647_chart_index = 1 }
    }
}

GDP647_format_chart_axis = {
    set_temp_variable = { GDP647_chart_axis_unit = 0 }
    if = {
        limit = { check_variable = { GDP647_chart_metric = 2 } }
        if = {
            limit = { check_variable = { var = GDP647_chart_axis value = 1000 compare = greater_than_or_equals } }
            divide_temp_variable = { GDP647_chart_axis = 1000 }
            set_temp_variable = { GDP647_chart_axis_unit = 1 }
        }
    }
    else_if = {
        limit = { check_variable = { var = GDP647_chart_axis value = 1000 compare = greater_than_or_equals } }
        divide_temp_variable = { GDP647_chart_axis = 1000 }
        set_temp_variable = { GDP647_chart_axis_unit = 3 }
    }
    else_if = {
        limit = { check_variable = { var = GDP647_chart_axis value = 1 compare = greater_than_or_equals } }
        set_temp_variable = { GDP647_chart_axis_unit = 2 }
    }
    else_if = {
        limit = { check_variable = { var = GDP647_chart_axis value = 0.001 compare = greater_than_or_equals } }
        multiply_temp_variable = { GDP647_chart_axis = 1000 }
        set_temp_variable = { GDP647_chart_axis_unit = 1 }
    }
    else = { multiply_temp_variable = { GDP647_chart_axis = 1000000 } }
}

# Any two consecutive month starts are at most 62 days apart.
# Starting from the first snapshot, 80 days selects the third monthly capture.
GDP647_prepare_chart_archive = {
    set_variable = { global.GDP647_chart_archive_due = 0 }
    set_temp_variable = { GDP647_chart_archive_elapsed = global.GDP647_capture_day }
    subtract_from_temp_variable = { GDP647_chart_archive_elapsed = global.GDP647_chart_last_archive_day }
    if = {
        limit = {
            OR = {
                NOT = { has_variable = global.GDP647_chart_last_archive_day }
                check_variable = { var = GDP647_chart_archive_elapsed value = 80 compare = greater_than_or_equals }
            }
        }
        set_variable = { global.GDP647_chart_archive_due = 1 }
        set_variable = { global.GDP647_chart_last_archive_day = global.GDP647_capture_day }
    }
}
