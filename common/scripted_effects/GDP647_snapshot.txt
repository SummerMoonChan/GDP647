GDP647_begin = {
    GDP647_capture_state_economy_categories = yes

# State scope; PREV is its controller at capture.
GDP647_compute_state = {
    set_variable = { GDP647_geo_U9_frozen = 1 }
    GDP647_geo_U9_output = yes
    set_variable = { GDP647_geo_base_gdp = GDP647_geo_agri }
    add_to_variable = { GDP647_geo_base_gdp = GDP647_geo_civ }
    add_to_variable = { GDP647_geo_base_gdp = GDP647_geo_mil }
    add_to_variable = { GDP647_geo_base_gdp = GDP647_geo_service }
    add_to_variable = { GDP647_geo_base_gdp = GDP647_geo_resources }
    set_variable = { GDP647_geo_SA_frozen_capital_base_m = GDP647_geo_base_gdp }
    subtract_from_variable = { GDP647_geo_SA_frozen_capital_base_m = GDP647_geo_service }
    add_to_variable = { GDP647_geo_SA_frozen_capital_base_m = GDP647_geo_SA_legacy_service }

    set_variable = { GDP647_geo_investment = GDP647_geo_invested_capital_m }
    multiply_variable = { GDP647_geo_investment = 0.2 }
    multiply_variable = { GDP647_geo_investment = GDP647_geo_operation }
    set_variable = { GDP647_geo_P5_next_state_total = GDP647_geo_base_gdp }
    add_to_variable = { GDP647_geo_P5_next_state_total = GDP647_geo_investment }
    set_variable = { GDP647_geo_P5_next_GDP_state_per_capita_v41 = 0 }
    if = {
        limit = { check_variable = { GDP647_geo_pop > 0 } }
        set_variable = { GDP647_geo_P5_next_GDP_state_per_capita_v41 = GDP647_geo_P5_next_state_total }
        divide_variable = { GDP647_geo_P5_next_GDP_state_per_capita_v41 = GDP647_geo_pop }
    }
    add_to_variable = { PREV.GDP647_geo_P5_next_GDP_total = GDP647_geo_P5_next_state_total }
    if = {
        limit = { check_variable = { GDP647_geo_P5_is_core = 1 } }
        add_to_variable = { PREV.GDP647_next_core_population = GDP647_geo_pop }
    }
    else = { add_to_variable = { PREV.GDP647_geo_P5_next_GDP_overseas_m = GDP647_geo_P5_next_state_total } }
}

    GDP647_prepare_growth_sample = yes
    GDP647_prepare_chart_archive = yes
    clear_array = global.GDP647_countries
    clear_array = global.GDP647_states
    for_each_scope_loop = {
        array = global.GDP647_sampled_countries
        GDP647_geo_SA_finish_frozen = yes
        set_temp_variable = { GDP647_prior_country_gdp = GDP647_country_gdp }
        set_variable = { GDP647_country_gdp = GDP647_geo_P5_next_GDP_total }
        set_variable = { GDP647_published_core_population = GDP647_next_core_population }
        set_variable = { GDP647_published_country_population = 0 }
        set_variable = { GDP647_country_people_k = 0 }
        set_variable = { GDP647_country_sector_agriculture = 0 }
        set_variable = { GDP647_country_sector_civilian = 0 }
        set_variable = { GDP647_country_sector_military = 0 }
        set_variable = { GDP647_country_sector_services = 0 }
        set_variable = { GDP647_country_sector_resources = 0 }
        set_variable = { GDP647_country_sector_reinvestment = 0 }
        set_temp_variable = { GDP647_prior_country_pc = GDP647_core_gdp_per_capita }
        set_variable = { GDP647_core_gdp_per_capita = 0 }
        if = {
            limit = { check_variable = { GDP647_next_core_population > 0 } }
            set_variable = { GDP647_core_gdp_per_capita = GDP647_country_gdp }
            subtract_from_variable = { GDP647_core_gdp_per_capita = GDP647_geo_P5_next_GDP_overseas_m }
            clamp_variable = { var = GDP647_core_gdp_per_capita min = 0 max = GDP647_country_gdp }
            divide_variable = { GDP647_core_gdp_per_capita = GDP647_next_core_population }
        }
        for_each_scope_loop = {
            array = GDP647_geo_P5_states
            set_temp_variable = { GDP647_growth_prior_published = GDP647_state_gdp }
            set_variable = { GDP647_state_gdp = GDP647_geo_P5_next_state_total }
            set_variable = { GDP647_state_sector_agriculture = GDP647_geo_agri }
            set_variable = { GDP647_state_sector_civilian = GDP647_geo_civ }
            set_variable = { GDP647_state_sector_military = GDP647_geo_mil }
            set_variable = { GDP647_state_sector_services = GDP647_geo_service }
            set_variable = { GDP647_state_sector_resources = GDP647_geo_resources }
            set_variable = { GDP647_state_sector_reinvestment = GDP647_geo_investment }
            set_variable = { GDP647_published_state_population = GDP647_geo_U9_f_pop }
            set_variable = { GDP647_state_population_10k = GDP647_published_state_population }
            multiply_variable = { GDP647_state_population_10k = 100 }
            if = {
                limit = { has_variable = GDP647_capture_population_k }
                set_variable = { GDP647_state_people_k = GDP647_capture_population_k }
            }
            else = {
                set_variable = { GDP647_state_people_k = GDP647_state_population_10k }
                multiply_variable = { GDP647_state_people_k = 10 }
            }
            add_to_variable = { PREV.GDP647_country_people_k = GDP647_state_people_k }
            GDP647_format_state_population = yes
            set_variable = { GDP647_published_state_is_core = GDP647_geo_P5_is_core }
            clear_variable = GDP647_state_gdp_ratio_pct
            if = {
                limit = { check_variable = { GDP647_published_state_is_core = 0 } }
                set_temp_variable = { GDP647_state_gdp_potential = GDP647_geo_invested_capital_m }
                multiply_temp_variable = { GDP647_state_gdp_potential = 0.2 }
                add_to_temp_variable = { GDP647_state_gdp_potential = GDP647_geo_population_potential }
                add_to_temp_variable = { GDP647_state_gdp_potential = GDP647_geo_industry_potential }
                add_to_temp_variable = { GDP647_state_gdp_potential = GDP647_geo_C92_resource_potential }
                if = {
                    limit = { check_variable = { GDP647_state_gdp_potential > 0 } }
                    set_variable = { GDP647_state_gdp_ratio_pct = GDP647_state_gdp }
                    divide_variable = { GDP647_state_gdp_ratio_pct = GDP647_state_gdp_potential }
                    multiply_variable = { GDP647_state_gdp_ratio_pct = 100 }
                }
            }
            add_to_variable = { PREV.GDP647_published_country_population = GDP647_published_state_population }
            add_to_variable = { PREV.GDP647_country_sector_agriculture = GDP647_state_sector_agriculture }
            add_to_variable = { PREV.GDP647_country_sector_civilian = GDP647_state_sector_civilian }
            add_to_variable = { PREV.GDP647_country_sector_military = GDP647_state_sector_military }
            add_to_variable = { PREV.GDP647_country_sector_services = GDP647_state_sector_services }
            add_to_variable = { PREV.GDP647_country_sector_resources = GDP647_state_sector_resources }
            add_to_variable = { PREV.GDP647_country_sector_reinvestment = GDP647_state_sector_reinvestment }
            set_temp_variable = { GDP647_pc_prior = GDP647_state_gdp_per_capita }
            set_variable = { GDP647_state_gdp_per_capita = GDP647_geo_P5_next_GDP_state_per_capita_v41 }
            add_to_array = { array = global.GDP647_states }
            if = {
                limit = {
                    NOT = { check_variable = { GDP647_growth_controller = GDP647_capture_controller } }
                }
                clear_variable = GDP647_gdp_growth_last_revision
            }
            set_variable = { GDP647_growth_controller = GDP647_capture_controller }
            set_temp_variable = { GDP647_growth_sample = GDP647_state_gdp }
            set_temp_variable = { GDP647_pc_current = GDP647_state_gdp_per_capita }
            GDP647_record_gdp_pc_growth = yes
            GDP647_record_gdp_growth = yes
            GDP647_format_state_amounts = yes
            GDP647_invest = yes
        }
        set_variable = { GDP647_country_population_10k = GDP647_published_country_population }
        multiply_variable = { GDP647_country_population_10k = 100 }
        GDP647_format_country_population = yes
        set_temp_variable = { GDP647_growth_prior_published = GDP647_prior_country_gdp }
        set_temp_variable = { GDP647_growth_sample = GDP647_country_gdp }
        set_temp_variable = { GDP647_pc_prior = GDP647_prior_country_pc }

        GDP647_record_gdp_growth = yes

        GDP647_refresh_diplomacy_scope = yes
        GDP647_format_country_amounts = yes
        if = { limit = { exists = yes } add_to_array = { array = global.GDP647_countries } }

    GDP647_refresh_faction_directory = yes

    GDP647_archive_world_chart = yes
    GDP647_rank_follow_countries = yes
    every_country = {

GDP647_invest = {
    # Add capital after publishing this month's GDP. The initial snapshot only starts the ledger.
    if = {
        limit = { has_state_flag = GDP647_geo_v4_capital_started }
        set_variable = { GDP647_geo_v4_depreciation = GDP647_geo_invested_capital_m }
        multiply_variable = { GDP647_geo_v4_depreciation = 0.03 }
        divide_variable = { GDP647_geo_v4_depreciation = 12 }
        subtract_from_variable = { GDP647_geo_invested_capital_m = GDP647_geo_v4_depreciation }

        set_variable = { GDP647_geo_v4_new_capital = GDP647_geo_SA_frozen_capital_base_m }
        if = {
            limit = { check_variable = { GDP647_geo_P5_f_war = 0 } }
            multiply_variable = { GDP647_geo_v4_new_capital = 0.015 }
        }
        else = { multiply_variable = { GDP647_geo_v4_new_capital = 0.0075 } }
        divide_variable = { GDP647_geo_v4_new_capital = 12 }
        add_to_variable = { GDP647_geo_invested_capital_m = GDP647_geo_v4_new_capital }
        clamp_variable = { var = GDP647_geo_invested_capital_m min = 0 }
    }
    else = { set_state_flag = GDP647_geo_v4_capital_started }
}

# Economic columns toggle direction. TAG and population also return to the last economic order.
GDP647_change_sort = {
    GDP647_load_sort_key = yes
    set_temp_variable = { GDP647_next_order_key = GDP647_requested_sort_key }
    set_temp_variable = { GDP647_opposite_key = GDP647_requested_sort_key }
    multiply_temp_variable = { GDP647_opposite_key = -1 }
    if = {
        limit = { check_variable = { GDP647_order_key = GDP647_requested_sort_key } }
        set_temp_variable = { GDP647_next_order_key = GDP647_opposite_key }
    }
    else_if = {
        limit = {
            check_variable = { GDP647_requested_sort_key > 2 }
            check_variable = { GDP647_order_key = GDP647_opposite_key }
        }
        set_temp_variable = { GDP647_next_order_key = GDP647_economic_sort_keys^GDP647_page_index }
        if = {
            limit = { check_variable = { GDP647_next_order_key = 0 } }
            set_temp_variable = { GDP647_next_order_key = 1 }
        }
    }
    if = {
        limit = { check_variable = { GDP647_requested_sort_key < 3 } }
        set_variable = { GDP647_economic_sort_keys^GDP647_page_index = GDP647_next_order_key }
    }
    else_if = {
        limit = {
            check_variable = { GDP647_order_key < 3 }
            check_variable = { GDP647_order_key > -3 }
        }
        set_variable = { GDP647_economic_sort_keys^GDP647_page_index = GDP647_order_key }
    }
    set_variable = { GDP647_page_sort_keys^GDP647_page_index = GDP647_next_order_key }
    GDP647_reset_list_page = yes
    GDP647_refresh_order = yes
    add_to_variable = { global.GDP647_ui_revision = 1 }
}

GDP647_load_sort_key = {
    set_temp_variable = { GDP647_page_index = 0 }
    if = { limit = { has_variable = GDP647_view } set_temp_variable = { GDP647_page_index = GDP647_view } }
    resize_array = { array = GDP647_page_sort_keys size = 3 value = 0 }
    resize_array = { array = GDP647_economic_sort_keys size = 3 value = 0 }
    set_temp_variable = { GDP647_order_key = GDP647_page_sort_keys^GDP647_page_index }
    if = {
        limit = { check_variable = { GDP647_order_key = 0 } }
        set_temp_variable = { GDP647_order_key = 1 }
        if = {
            limit = { check_variable = { GDP647_page_index = 0 } }
            if = { limit = { check_variable = { GDP647_sort_key = 2 } } set_temp_variable = { GDP647_order_key = 2 } }
            if = { limit = { has_country_flag = GDP647_sort_ascending } multiply_temp_variable = { GDP647_order_key = -1 } }
        }
    }
}

# The rank can keep the last economic metric while another column orders the rows.
GDP647_load_rank_basis = {
    set_temp_variable = { GDP647_rank_basis = GDP647_order_metric }
    if = {
        limit = { check_variable = { GDP647_order_metric > 2 } }
        set_temp_variable = { GDP647_rank_basis = GDP647_economic_sort_keys^GDP647_page_index }
        if = {
            limit = { check_variable = { GDP647_rank_basis < 0 } }
            multiply_temp_variable = { GDP647_rank_basis = -1 }
        }
        clamp_temp_variable = { var = GDP647_rank_basis min = 1 max = 2 }
    }
}

GDP647_refresh_order = {
    if = {
        limit = {
            check_variable = { GDP647_view = 1 }
            OR = {
                NOT = { has_global_flag = GDP647_live_faction_directory_ready }
                any_of_scopes = { array = global.GDP647_factions NOT = { has_variable = GDP647_faction_core_gdp_per_capita_compact_unit } }
                has_global_flag = GDP647_faction_directory_dirty
                any_of_scopes = {
                    array = global.GDP647_factions
                    OR = { exists = no is_in_faction = no is_faction_leader = no NOT = { has_variable = GDP647_faction_sector_agriculture } }
                }
                any_of_scopes = {
                    array = global.GDP647_countries
                    exists = yes
                    is_in_faction = yes
                    is_faction_leader = yes
                    NOT = { is_in_array = { array = global.GDP647_factions value = THIS } }
                }
            }
        }
        set_temp_variable = { GDP647_record_faction_snapshot = 0 }
        GDP647_refresh_faction_directory = yes
    }
    if = {
        limit = {
            NOT = { check_variable = { GDP647_view > 0 } }
            NOT = { has_global_flag = GDP647_world_totals_ready }
        }
        GDP647_publish_economic_benchmarks = yes
    }
    GDP647_load_sort_key = yes
    set_temp_variable = { GDP647_order_metric = GDP647_order_key }
    if = {
        limit = { check_variable = { GDP647_order_metric < 0 } }
        multiply_temp_variable = { GDP647_order_metric = -1 }
    }
    GDP647_load_rank_basis = yes
    set_variable = { GDP647_name_sort_frame = 1 }
    set_variable = { GDP647_population_sort_frame = 1 }
    if = {
        limit = { check_variable = { GDP647_order_metric = 3 } }
        set_variable = { GDP647_name_sort_frame = 2 }
    }
    else_if = {
        limit = { check_variable = { GDP647_order_metric = 4 } }
        set_variable = { GDP647_population_sort_frame = 2 }
    }
    set_variable = { GDP647_rank_per_capita_frame = GDP647_rank_basis }
    set_variable = { GDP647_rank_total_frame = 3 }
    subtract_from_variable = { GDP647_rank_total_frame = GDP647_rank_per_capita_frame }
    set_variable = { GDP647_population_filter_frame = 1 }
    if = {
        limit = {
            OR = {
                AND = { check_variable = { GDP647_page_index = 0 } has_country_flag = GDP647_country_show_small_population }
                AND = { check_variable = { GDP647_page_index = 2 } has_country_flag = GDP647_state_show_small_population }
            }
        }
        set_variable = { GDP647_population_filter_frame = 2 }
    }
    clr_country_flag = GDP647_per_capita_order
    set_temp_variable = { GDP647_minimum_population = 0 }
    if = {
        limit = { check_variable = { GDP647_rank_basis = 2 } }
        set_country_flag = GDP647_per_capita_order
        if = {
            limit = { check_variable = { GDP647_page_index = 2 } NOT = { has_country_flag = GDP647_state_show_small_population } }
            set_temp_variable = { GDP647_minimum_population = @GDP647_min_state_population }
        }
        else_if = {
            limit = { check_variable = { GDP647_page_index = 0 } NOT = { has_country_flag = GDP647_country_show_small_population } }
            set_temp_variable = { GDP647_minimum_population = @GDP647_min_country_population }
        }
    }
    if = {
        limit = { check_variable = { GDP647_order_metric = 3 } NOT = { has_global_flag = GDP647_tag_table_ready } }
        GDP647_define_tag_order = yes
        set_global_flag = GDP647_tag_table_ready
    }
    if = {
        limit = { check_variable = { GDP647_page_index = 2 } }
        GDP647_refresh_state_ranking = yes
    }
    else = { GDP647_rebuild_current_order = yes }
    GDP647_load_list_page = yes
}

GDP647_rebuild_current_order = {
    clear_array = GDP647_economic_order
    if = {
        limit = { check_variable = { GDP647_order_metric > 2 } }
        GDP647_build_economic_ranks = yes
    }
    GDP647_build_order = yes
    clear_array = GDP647_row_ranks
    set_temp_variable = { GDP647_row_position = 0 }
    set_temp_variable = { GDP647_row_count = GDP647_display_order^num }
    for_each_scope_loop = {
        array = GDP647_display_order
        add_to_temp_variable = { GDP647_row_position = 1 }
        set_temp_variable = { GDP647_row_rank = GDP647_row_position }
        if = {
            limit = { check_variable = { GDP647_order_metric > 2 } }
            set_temp_variable = { GDP647_row_rank = GDP647_economic_rank }
        }
        else_if = {
            limit = { check_variable = { GDP647_order_key < 0 } }
            set_temp_variable = { GDP647_row_rank = GDP647_row_count }
            subtract_from_temp_variable = { GDP647_row_rank = GDP647_row_position }
            add_to_temp_variable = { GDP647_row_rank = 1 }
        }
        add_to_array = { array = PREV.GDP647_row_ranks value = GDP647_row_rank }
    }
}

GDP647_build_order = {
    clear_array = GDP647_display_order
    clear_temp_array = GDP647_rank_scores
    if = {
        limit = { check_variable = { GDP647_page_index = 1 } }
        for_each_scope_loop = {
            array = global.GDP647_factions
            if = {
                limit = { check_variable = { GDP647_order_metric = 3 } }
                GDP647_tag_sort_score = yes
            }
            PREV = {
                if = {
                    limit = { NOT = { check_variable = { GDP647_order_metric = 3 } } }
                    set_temp_variable = { GDP647_rank_score = PREV.GDP647_faction_gdp }
                }
                if = {
                    limit = { check_variable = { GDP647_order_metric = 4 } }
                    set_temp_variable = { GDP647_rank_score = PREV.GDP647_faction_population_10k }
                }
                else_if = {
                    limit = { check_variable = { GDP647_order_metric = 2 } }
                    set_temp_variable = { GDP647_rank_score = PREV.GDP647_faction_core_gdp_per_capita }
                }
                GDP647_insert_row = yes
            }
        }
    }
    else_if = {
        limit = { check_variable = { GDP647_page_index = 2 } }
        for_each_scope_loop = {
            array = global.GDP647_states
            if = {
                limit = {
                    OR = {
                        NOT = { has_variable = GDP647_published_state_population }
                        check_variable = {
                            var = GDP647_published_state_population
                            value = GDP647_minimum_population
                            compare = greater_than_or_equals
                        }
                    }
                }
                if = {
                    limit = { check_variable = { GDP647_order_metric = 3 } }
                    set_temp_variable = { GDP647_rank_score = -46657 }
                    controller = { GDP647_tag_sort_score = yes }
                }
                PREV = {
                    if = {
                        limit = { NOT = { check_variable = { GDP647_order_metric = 3 } } }
                        set_temp_variable = { GDP647_rank_score = PREV.GDP647_state_gdp }
                    }
                    if = {
                        limit = { check_variable = { GDP647_order_metric = 4 } }
                        set_temp_variable = { GDP647_rank_score = PREV.GDP647_state_population_10k }
                    }
                    if = {
                        limit = { check_variable = { GDP647_order_metric = 2 } }
                        set_temp_variable = { GDP647_rank_score = PREV.GDP647_state_gdp_per_capita }
                    }
                    GDP647_insert_row = yes
                }
            }
        }
    }
    else = {
        for_each_scope_loop = {
            array = global.GDP647_countries
            if = {
                limit = {
                    OR = {
                        NOT = { has_variable = GDP647_published_core_population }
                        check_variable = {
                            var = GDP647_published_core_population
                            value = GDP647_minimum_population
                            compare = greater_than_or_equals
                        }
                    }
                }
                if = {
                    limit = { check_variable = { GDP647_order_metric = 3 } }
                    GDP647_tag_sort_score = yes
                }
                PREV = {
                    if = {
                        limit = { NOT = { check_variable = { GDP647_order_metric = 3 } } }
                        set_temp_variable = { GDP647_rank_score = PREV.GDP647_country_gdp }
                    }
                    if = {
                        limit = { check_variable = { GDP647_order_metric = 4 } }
                        set_temp_variable = { GDP647_rank_score = PREV.GDP647_country_population_10k }
                    }
                    if = {
                        limit = { check_variable = { GDP647_order_metric = 2 } }
                        set_temp_variable = { GDP647_rank_score = PREV.GDP647_core_gdp_per_capita }
                    }
                    GDP647_insert_row = yes
                }
            }
        }
    }
}

# Find the insertion index with a binary search. Ordered ties retain snapshot order.
GDP647_find_rank_index = {
    if = {
        limit = { check_variable = { GDP647_order_key < 0 } }
        multiply_temp_variable = { GDP647_rank_score = -1 }
    }
    set_temp_variable = { GDP647_rank_low = 0 }
    set_temp_variable = { GDP647_rank_high = GDP647_rank_scores^num }
    while_loop_effect = {
        limit = { check_variable = { var = GDP647_rank_low value = GDP647_rank_high compare = less_than } }
        set_temp_variable = { GDP647_rank_middle = GDP647_rank_low }
        add_to_temp_variable = { GDP647_rank_middle = GDP647_rank_high }
        divide_temp_variable = { GDP647_rank_middle = 2 }
        round_temp_variable = GDP647_rank_middle
        set_temp_variable = { GDP647_rank_last = GDP647_rank_high }
        subtract_from_temp_variable = { GDP647_rank_last = 1 }
        clamp_temp_variable = { var = GDP647_rank_middle min = GDP647_rank_low max = GDP647_rank_last }
        if = {
            limit = {
                check_variable = { var = GDP647_rank_scores^GDP647_rank_middle value = GDP647_rank_score compare = greater_than_or_equals }
            }
            set_temp_variable = { GDP647_rank_low = GDP647_rank_middle }
            add_to_temp_variable = { GDP647_rank_low = 1 }
        }
        else = { set_temp_variable = { GDP647_rank_high = GDP647_rank_middle } }
    }
}

GDP647_insert_row = {
    GDP647_find_rank_index = yes
    add_to_array = { array = GDP647_display_order value = PREV index = GDP647_rank_low }
    add_to_temp_array = { array = GDP647_rank_scores value = GDP647_rank_score index = GDP647_rank_low }
}

# Score TAGs in reverse numeric order because the common sorter starts with the highest score.
GDP647_tag_sort_score = {
    if = {
        limit = { NOT = { has_variable = GDP647_tag_alphabetic_index } }
        meta_effect = {
            text = { set_variable = { GDP647_tag_alphabetic_index = global.GDP647_tag_[TAG] } }
            TAG = "[THIS.GetTag]"
        }
        if = {
            limit = { NOT = { check_variable = { GDP647_tag_alphabetic_index > 0 } } }
            set_variable = { GDP647_tag_alphabetic_index = 46657 }
        }
    }
    set_temp_variable = { GDP647_rank_score = GDP647_tag_alphabetic_index }
    multiply_temp_variable = { GDP647_rank_score = -1 }
}

# The shared page holds only the rows being drawn. State ranks may come from
# a cached snapshot rather than the active tab's shared sorting workspace.
GDP647_load_list_page = {
    GDP647_load_sort_key = yes
    resize_array = { array = GDP647_page_offsets size = 3 value = 0 }
    set_temp_variable = { GDP647_page_row_count = GDP647_display_order^num }
    if = {
        limit = { check_variable = { GDP647_page_index = 2 } }
        set_temp_variable = { GDP647_page_row_count = GDP647_state_cached_order^num }
    }
    clear_temp_array = GDP647_filtered_indices
    set_temp_variable = { GDP647_source_index = 0 }
    set_temp_variable = { GDP647_filter_active = 0 }
    if = {
        limit = {
            check_variable = { GDP647_page_index = 0 }
            check_variable = { GDP647_faction_filter^num > 0 }
        }
        set_temp_variable = { GDP647_filter_active = 1 }
        set_temp_variable = { GDP647_filter_target = GDP647_faction_filter^0 }
        for_each_scope_loop = {
            array = GDP647_display_order
            if = {
                limit = {
                    is_in_faction = yes
                    OR = { tag = var:GDP647_filter_target is_in_faction_with = var:GDP647_filter_target }
                }
                add_to_temp_array = { array = GDP647_filtered_indices value = GDP647_source_index }
            }
            add_to_temp_variable = { GDP647_source_index = 1 }
        }
        set_temp_variable = { GDP647_page_row_count = GDP647_filtered_indices^num }
    }
    else_if = {
        limit = {
            check_variable = { GDP647_page_index = 2 }
            check_variable = { GDP647_state_filter^num > 0 }
        }
        set_temp_variable = { GDP647_filter_active = 1 }
        set_temp_variable = { GDP647_filter_target = GDP647_state_filter^0 }
        for_each_scope_loop = {
            array = GDP647_state_cached_order
            if = {
                limit = { is_controlled_by = var:GDP647_filter_target }
                add_to_temp_array = { array = GDP647_filtered_indices value = GDP647_source_index }
            }
            add_to_temp_variable = { GDP647_source_index = 1 }
        }
        set_temp_variable = { GDP647_page_row_count = GDP647_filtered_indices^num }
    }
    set_temp_variable = { GDP647_last_page_offset = GDP647_page_row_count }
    subtract_from_temp_variable = { GDP647_last_page_offset = 1 }
    clamp_temp_variable = { var = GDP647_last_page_offset min = 0 }
    set_temp_variable = { GDP647_page_remainder = GDP647_last_page_offset }
    modulo_temp_variable = { var = GDP647_page_remainder value = @GDP647_page_size }
    subtract_from_temp_variable = { GDP647_last_page_offset = GDP647_page_remainder }
    set_temp_variable = { GDP647_current_page_offset = GDP647_page_offsets^GDP647_page_index }
    clamp_temp_variable = { var = GDP647_current_page_offset min = 0 max = GDP647_last_page_offset }
    # A save may still contain an offset from the former eleven-row pagination.
    set_temp_variable = { GDP647_current_page_remainder = GDP647_current_page_offset }
    modulo_temp_variable = { var = GDP647_current_page_remainder value = @GDP647_page_size }
    subtract_from_temp_variable = { GDP647_current_page_offset = GDP647_current_page_remainder }
    set_variable = { GDP647_page_offsets^GDP647_page_index = GDP647_current_page_offset }

    set_variable = { GDP647_page_number = GDP647_page_offsets^GDP647_page_index }
    divide_variable = { GDP647_page_number = @GDP647_page_size }
    add_to_variable = { GDP647_page_number = 1 }
    set_variable = { GDP647_page_count = GDP647_last_page_offset }
    divide_variable = { GDP647_page_count = @GDP647_page_size }
    add_to_variable = { GDP647_page_count = 1 }

    clear_array = GDP647_page_rows
    clear_array = GDP647_page_ranks
    set_temp_variable = { GDP647_page_cursor = GDP647_page_offsets^GDP647_page_index }
    set_temp_variable = { GDP647_page_end = GDP647_page_cursor }
    add_to_temp_variable = { GDP647_page_end = @GDP647_page_size }
    clamp_temp_variable = { var = GDP647_page_end max = GDP647_page_row_count }
    while_loop_effect = {
        limit = {
            check_variable = { GDP647_page_cursor < GDP647_page_end }
        }
        set_temp_variable = { GDP647_source_index = GDP647_page_cursor }
        if = {
            limit = { check_variable = { GDP647_filter_active = 1 } }
            set_temp_variable = { GDP647_source_index = GDP647_filtered_indices^GDP647_page_cursor }
        }
        if = {
            limit = { check_variable = { GDP647_page_index = 2 } }
            add_to_array = { array = GDP647_page_rows value = GDP647_state_cached_order^GDP647_source_index }
            add_to_array = { array = GDP647_page_ranks value = GDP647_state_cached_ranks^GDP647_source_index }
        }
        else = {
            add_to_array = { array = GDP647_page_rows value = GDP647_display_order^GDP647_source_index }
            add_to_array = { array = GDP647_page_ranks value = GDP647_row_ranks^GDP647_source_index }
        }
        add_to_temp_variable = { GDP647_page_cursor = 1 }
    }
}
