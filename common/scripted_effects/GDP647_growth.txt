GDP647_prepare_growth_sample = {
    set_variable = { global.GDP647_growth_previous_revision = -1 }
    if = {
        limit = { has_variable = global.GDP647_growth_last_capture_day }
        set_temp_variable = { GDP647_growth_elapsed_days = global.GDP647_capture_day }
        subtract_from_temp_variable = { GDP647_growth_elapsed_days = global.GDP647_growth_last_capture_day }
        if = {
            limit = {
                check_variable = { var = GDP647_growth_elapsed_days value = 28 compare = greater_than_or_equals }
                check_variable = { var = GDP647_growth_elapsed_days value = 31 compare = less_than_or_equals }
            }
            set_variable = { global.GDP647_growth_previous_revision = global.GDP647_published_revision }
        }
    }
    set_variable = { global.GDP647_growth_last_capture_day = global.GDP647_capture_day }
    add_to_variable = { global.GDP647_published_revision = 1 }
}

# Newest first: index 0 is last month, and index 11 is this month last year.
GDP647_record_gdp_growth = {
    if = {
        limit = {
            OR = {
                NOT = { has_variable = GDP647_gdp_growth_last_revision }
                NOT = { check_variable = { GDP647_gdp_growth_last_revision = global.GDP647_growth_previous_revision } }
            }
        }
        clear_array = GDP647_gdp_growth_monthly_gdp
    }
    else_if = {
        limit = {
            check_variable = { GDP647_gdp_growth_monthly_gdp^num = 0 }
            check_variable = { GDP647_growth_prior_published > 0 }
        }
        # Seed one known-good monthly value when upgrading an existing save.
        add_to_array = { array = GDP647_gdp_growth_monthly_gdp value = GDP647_growth_prior_published }
    }
    set_variable = { GDP647_gdp_growth_mom_valid = 0 }
    if = {
        limit = {
            check_variable = { var = GDP647_gdp_growth_monthly_gdp^num value = 1 compare = greater_than_or_equals }
            check_variable = { GDP647_gdp_growth_monthly_gdp^0 > 0 }
        }
        set_variable = { GDP647_gdp_growth_mom = GDP647_growth_sample }
        divide_variable = { GDP647_gdp_growth_mom = GDP647_gdp_growth_monthly_gdp^0 }
        subtract_from_variable = { GDP647_gdp_growth_mom = 1 }
        multiply_variable = { GDP647_gdp_growth_mom = 100 }
        set_variable = { GDP647_gdp_growth_mom_valid = 1 }
    }
    set_variable = { GDP647_gdp_growth_yoy_valid = 0 }
    if = {
        limit = {
            check_variable = { var = GDP647_gdp_growth_monthly_gdp^num value = 12 compare = greater_than_or_equals }
            check_variable = { GDP647_gdp_growth_monthly_gdp^11 > 0 }
        }
        set_variable = { GDP647_gdp_growth_yoy = GDP647_growth_sample }
        divide_variable = { GDP647_gdp_growth_yoy = GDP647_gdp_growth_monthly_gdp^11 }
        subtract_from_variable = { GDP647_gdp_growth_yoy = 1 }
        multiply_variable = { GDP647_gdp_growth_yoy = 100 }
        set_variable = { GDP647_gdp_growth_yoy_valid = 1 }
    }
    add_to_array = { array = GDP647_gdp_growth_monthly_gdp value = GDP647_growth_sample index = 0 }
    if = {
        limit = { check_variable = { GDP647_gdp_growth_monthly_gdp^num > 12 } }
        remove_from_array = { array = GDP647_gdp_growth_monthly_gdp index = 12 }
    }
    set_variable = { GDP647_gdp_growth_last_revision = global.GDP647_published_revision }
}
