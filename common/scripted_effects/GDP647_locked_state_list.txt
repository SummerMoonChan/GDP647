# Keep the player's existing world sort, grouped into contiguous controller slices.
GDP647_build_locked_state_orders = {
    clear_array = GDP647_locked_state_counts
    clear_array = GDP647_locked_state_starts
    clear_array = GDP647_locked_state_rows
    clear_array = GDP647_locked_state_ranks
    set_temp_variable = { GDP647_locked_country_slots = 0 }
    for_each_scope_loop = {
        array = global.GDP647_countries
        set_temp_variable = { GDP647_locked_next_country_slot = id }
        add_to_temp_variable = { GDP647_locked_next_country_slot = 1 }
        clamp_temp_variable = { var = GDP647_locked_country_slots min = GDP647_locked_next_country_slot }
    }
    resize_array = { array = GDP647_locked_state_counts size = GDP647_locked_country_slots }
    resize_array = { array = GDP647_locked_state_starts size = GDP647_locked_country_slots }
    resize_array = { array = GDP647_locked_page_offsets size = GDP647_locked_country_slots }
    set_temp_variable = { GDP647_locked_rank_slots = 0 }
    for_each_scope_loop = {
        array = GDP647_state_cached_order
        set_temp_variable = { GDP647_locked_next_rank_slot = id }
        add_to_temp_variable = { GDP647_locked_next_rank_slot = 1 }
        clamp_temp_variable = { var = GDP647_locked_rank_slots min = GDP647_locked_next_rank_slot }
        controller = {
            set_temp_variable = { GDP647_locked_controller_id = id }
            PREV.PREV = { add_to_variable = { GDP647_locked_state_counts^GDP647_locked_controller_id = 1 } }
        }
    }
    set_temp_variable = { GDP647_locked_state_start = 0 }
    clear_temp_array = GDP647_locked_state_write_indices
    resize_temp_array = { array = GDP647_locked_state_write_indices size = GDP647_locked_country_slots }
    for_each_scope_loop = {
        array = global.GDP647_countries
        set_temp_variable = { GDP647_locked_controller_id = id }
        PREV = {
            if = {
                limit = { check_variable = { GDP647_locked_state_counts^GDP647_locked_controller_id > 0 } }
                set_variable = { GDP647_locked_state_starts^GDP647_locked_controller_id = GDP647_locked_state_start }
                set_temp_variable = { GDP647_locked_state_write_indices^GDP647_locked_controller_id = GDP647_locked_state_start }
                add_to_temp_variable = { GDP647_locked_state_start = GDP647_locked_state_counts^GDP647_locked_controller_id }
            }
        }
    }
    resize_array = { array = GDP647_locked_state_rows size = GDP647_locked_state_start }
    resize_array = { array = GDP647_locked_state_ranks size = GDP647_locked_rank_slots }
    set_temp_variable = { GDP647_locked_world_index = 0 }
    for_each_scope_loop = {
        array = GDP647_state_cached_order
        set_temp_variable = { GDP647_locked_row_state = THIS }
        set_temp_variable = { GDP647_locked_row_id = id }
        controller = {
            set_temp_variable = { GDP647_locked_controller_id = id }
            set_temp_variable = { GDP647_locked_write_index = GDP647_locked_state_write_indices^GDP647_locked_controller_id }
            PREV.PREV = {
                set_variable = { GDP647_locked_state_rows^GDP647_locked_write_index = GDP647_locked_row_state }
                set_variable = { GDP647_locked_state_ranks^GDP647_locked_row_id = GDP647_state_cached_ranks^GDP647_locked_world_index }
            }
            add_to_temp_variable = { GDP647_locked_state_write_indices^GDP647_locked_controller_id = 1 }
        }
        add_to_temp_variable = { GDP647_locked_world_index = 1 }
    }
    set_country_flag = GDP647_locked_state_orders_ready
}

GDP647_change_locked_state_page = {
    controller = { set_temp_variable = { GDP647_locked_controller_id = id } }
    set_temp_variable = { GDP647_locked_count = ROOT.GDP647_locked_state_counts^GDP647_locked_controller_id }
    set_temp_variable = { GDP647_locked_offset = ROOT.GDP647_locked_page_offsets^GDP647_locked_controller_id }
    set_temp_variable = { GDP647_locked_last_offset = GDP647_locked_count }
    subtract_from_temp_variable = { GDP647_locked_last_offset = 1 }
    clamp_temp_variable = { var = GDP647_locked_last_offset min = 0 }
    set_temp_variable = { GDP647_locked_remainder = GDP647_locked_last_offset }
    modulo_temp_variable = { var = GDP647_locked_remainder value = 10 }
    subtract_from_temp_variable = { GDP647_locked_last_offset = GDP647_locked_remainder }
    clamp_temp_variable = { var = GDP647_locked_offset min = 0 max = GDP647_locked_last_offset }
    multiply_temp_variable = { GDP647_locked_page_delta = 10 }
    add_to_temp_variable = { GDP647_locked_offset = GDP647_locked_page_delta }
    clamp_temp_variable = { var = GDP647_locked_offset min = 0 max = GDP647_locked_last_offset }
    ROOT = {
        if = {
            limit = { check_variable = { GDP647_locked_page_offsets^num = 0 } }
            resize_array = { array = GDP647_locked_page_offsets size = GDP647_locked_state_counts^num }
        }
        set_variable = { GDP647_locked_page_offsets^GDP647_locked_controller_id = GDP647_locked_offset }
    }
}
