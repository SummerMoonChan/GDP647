GDP647_pin_country_row = {
        ROOT = {
            set_country_flag = GDP647_manual_selection
            clear_array = GDP647_inspected_country
            add_to_array = { array = GDP647_inspected_country value = PREV }
            GDP647_select_state_for_country = yes

            GDP647_refresh_country_summary = yes
            if = {
                limit = { has_country_flag = GDP647_country_own_faction }
                GDP647_reset_list_page = yes
                GDP647_load_list_page = yes
            }
            add_to_variable = { global.GDP647_ui_revision = 1 }
        }
}

# Build contiguous slices of the world ranking without re-sorting or renumbering.
GDP647_prepare_faction_country_rows = {
    if = {
        limit = {
            OR = {
                NOT = { has_global_flag = GDP647_faction_slots_ready }
                has_global_flag = GDP647_faction_directory_dirty
            }
        }
        set_temp_variable = { GDP647_record_faction_snapshot = 0 }
        GDP647_refresh_faction_directory = yes
    }
    set_temp_variable = { GDP647_faction_slot_count = global.GDP647_factions^num }
    add_to_temp_variable = { GDP647_faction_slot_count = 1 }
    clear_array = GDP647_faction_row_counts
    clear_array = GDP647_faction_row_starts
    clear_array = GDP647_faction_sorted_rows
    clear_array = GDP647_faction_sorted_ranks
    resize_array = { array = GDP647_faction_row_counts size = GDP647_faction_slot_count }
    resize_array = { array = GDP647_faction_row_starts size = GDP647_faction_slot_count }
    for_each_scope_loop = {
        array = GDP647_display_order
        if = {
            limit = { is_in_faction = yes check_variable = { GDP647_faction_member_slot > 0 } }
            set_temp_variable = { GDP647_faction_slot = GDP647_faction_member_slot }
            PREV = { add_to_variable = { GDP647_faction_row_counts^GDP647_faction_slot = 1 } }
        }
    }
    clear_temp_array = GDP647_faction_write_positions
    resize_temp_array = { array = GDP647_faction_write_positions size = GDP647_faction_slot_count }
    set_temp_variable = { GDP647_faction_slot = 1 }
    set_temp_variable = { GDP647_faction_row_end = 0 }
    while_loop_effect = {
        limit = { check_variable = { GDP647_faction_slot < GDP647_faction_slot_count } }
        set_variable = { GDP647_faction_row_starts^GDP647_faction_slot = GDP647_faction_row_end }
        set_temp_variable = { GDP647_faction_write_positions^GDP647_faction_slot = GDP647_faction_row_end }
        add_to_temp_variable = { GDP647_faction_row_end = GDP647_faction_row_counts^GDP647_faction_slot }
        add_to_temp_variable = { GDP647_faction_slot = 1 }
    }
    resize_array = { array = GDP647_faction_sorted_rows size = GDP647_faction_row_end }
    resize_array = { array = GDP647_faction_sorted_ranks size = GDP647_faction_row_end }
    set_temp_variable = { GDP647_faction_world_row = 0 }
    for_each_scope_loop = {
        array = GDP647_display_order
        if = {
            limit = { is_in_faction = yes check_variable = { GDP647_faction_member_slot > 0 } }
            set_temp_variable = { GDP647_faction_slot = GDP647_faction_member_slot }
            set_temp_variable = { GDP647_faction_write_at = GDP647_faction_write_positions^GDP647_faction_slot }
            set_temp_variable = { GDP647_faction_country_to_write = THIS }
            PREV = {
                set_variable = { GDP647_faction_sorted_rows^GDP647_faction_write_at = GDP647_faction_country_to_write }
                set_variable = { GDP647_faction_sorted_ranks^GDP647_faction_write_at = GDP647_row_ranks^GDP647_faction_world_row }
            }
            add_to_temp_variable = { GDP647_faction_write_positions^GDP647_faction_slot = 1 }
        }
        add_to_temp_variable = { GDP647_faction_world_row = 1 }
    }
}

GDP647_step_faction_country_rows = {
    set_temp_variable = { GDP647_native_faction_slot = 0 }
    if = {
        limit = { is_in_faction = yes }
        set_temp_variable = { GDP647_native_faction_slot = GDP647_faction_member_slot }
    }
    set_temp_variable = { GDP647_faction_native_count = ROOT.GDP647_faction_row_counts^GDP647_native_faction_slot }
    set_temp_variable = { GDP647_faction_native_start = ROOT.GDP647_faction_row_starts^GDP647_native_faction_slot }
    set_temp_variable = { GDP647_faction_native_last = GDP647_faction_native_count }
    subtract_from_temp_variable = { GDP647_faction_native_last = 1 }
    clamp_temp_variable = { var = GDP647_faction_native_last min = 0 }
    set_temp_variable = { GDP647_faction_native_mod = GDP647_faction_native_last }
    modulo_temp_variable = { var = GDP647_faction_native_mod value = 10 }
    subtract_from_temp_variable = { GDP647_faction_native_last = GDP647_faction_native_mod }
    set_temp_variable = { GDP647_faction_native_offset = ROOT.GDP647_page_offsets^0 }
    clamp_temp_variable = { var = GDP647_faction_native_offset min = 0 max = GDP647_faction_native_last }
    multiply_temp_variable = { GDP647_faction_page_step = 10 }
    add_to_temp_variable = { GDP647_faction_page_step = GDP647_faction_native_offset }
    clamp_temp_variable = { var = GDP647_faction_page_step min = 0 max = GDP647_faction_native_last }
    ROOT = {
        set_variable = { GDP647_page_offsets^0 = GDP647_faction_page_step }
        add_to_variable = { global.GDP647_ui_revision = 1 }
    }
}
