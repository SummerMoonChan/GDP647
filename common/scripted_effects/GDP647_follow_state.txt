# Diplomacy uses the selected country directly. These arrays serve its chart and faction cards.
GDP647_refresh_diplomacy_scope = {
    clear_array = GDP647_diplomacy_country
    add_to_array = { array = GDP647_diplomacy_country }
    clear_array = GDP647_diplomacy_faction
    if = {
        limit = { is_in_faction = yes is_faction_leader = yes }
        add_to_array = { array = GDP647_diplomacy_faction }
    }
    else_if = {
        limit = { is_in_faction = yes }
        faction_leader = { add_to_array = { array = PREV.GDP647_diplomacy_faction } }
    }
}

GDP647_initialize_follow_cache = {
    if = {
        limit = { NOT = { has_global_flag = GDP647_diplomacy_cache_ready } }
        for_each_scope_loop = {
            array = global.GDP647_countries
            GDP647_prepare_map_chart = yes
            GDP647_refresh_diplomacy_scope = yes
        }
        GDP647_rank_follow_countries = yes
        set_global_flag = GDP647_diplomacy_cache_ready
    }
}

GDP647_rank_follow_countries = {
    clear_array = global.GDP647_follow_world_order
    clear_temp_array = GDP647_rank_scores
    set_temp_variable = { GDP647_order_key = 1 }
    for_each_scope_loop = {
        array = global.GDP647_countries
        set_temp_variable = { GDP647_rank_score = GDP647_country_gdp }
        GDP647_find_rank_index = yes
        add_to_array = { array = global.GDP647_follow_world_order index = GDP647_rank_low }
        add_to_temp_array = { array = GDP647_rank_scores value = GDP647_rank_score index = GDP647_rank_low }
    }
    set_temp_variable = { GDP647_follow_rank = 0 }
    for_each_scope_loop = {
        array = global.GDP647_follow_world_order
        add_to_temp_variable = { GDP647_follow_rank = 1 }
        set_variable = { GDP647_follow_country_rank = GDP647_follow_rank }
    }
}

GDP647_prepare_map_chart = {
    set_variable = { GDP647_map_chart_count = GDP647_chart_total_archive^num }
    clamp_variable = { var = GDP647_map_chart_count max = 24 }
    if = {
        limit = { check_variable = { GDP647_map_chart_count > 1 } }
        GDP647_cache_map_chart_total = yes
        GDP647_cache_map_chart_pc = yes
    }
}

GDP647_render_map_chart_series = {
    clear_array = GDP647_chart_display_samples
    clear_array = GDP647_chart_plot_points
    set_variable = { GDP647_chart_count = GDP647_map_chart_count }
    set_temp_variable = { GDP647_map_plot_last = GDP647_chart_total_archive^num }
    subtract_from_temp_variable = { GDP647_map_plot_last = 1 }
    set_temp_variable = { GDP647_map_plot_span = GDP647_map_chart_count }
    subtract_from_temp_variable = { GDP647_map_plot_span = 1 }
    set_temp_variable = { GDP647_map_plot_i = 0 }
    while_loop_effect = {
        limit = { check_variable = { GDP647_map_plot_i < GDP647_map_chart_count } }
        set_temp_variable = { GDP647_map_plot_index = GDP647_map_plot_i }
        multiply_temp_variable = { GDP647_map_plot_index = GDP647_map_plot_last }
        divide_temp_variable = { GDP647_map_plot_index = GDP647_map_plot_span }
        round_temp_variable = GDP647_map_plot_index
        set_temp_variable = { GDP647_map_plot_source = GDP647_map_plot_last }
        subtract_from_temp_variable = { GDP647_map_plot_source = GDP647_map_plot_index }
        set_temp_variable = { GDP647_map_plot_value = GDP647_chart_total_archive^GDP647_map_plot_source }
        if = {
            limit = { check_variable = { GDP647_chart_scaler_metric = 2 } }
            set_temp_variable = { GDP647_map_plot_value = GDP647_chart_pc_archive^GDP647_map_plot_source }
        }
        add_to_array = { array = GDP647_chart_display_samples value = GDP647_map_plot_value }
        add_to_temp_variable = { GDP647_map_plot_i = 1 }
    }
    GDP647_scale_chart_samples = yes
}

# Native selected-state cards read the controller and faction without changing the player's selection.
GDP647_refresh_state_follow_scope = {
    clear_array = GDP647_state_follow_country_scope
    clear_array = GDP647_state_follow_faction_scope
    controller = {
        add_to_array = { array = PREV.GDP647_state_follow_country_scope }
        if = {
            limit = { is_in_faction = yes is_faction_leader = yes }
            add_to_array = { array = PREV.GDP647_state_follow_faction_scope }
        }
        else_if = {
            limit = { is_in_faction = yes }
            faction_leader = { add_to_array = { array = PREV.PREV.GDP647_state_follow_faction_scope } }
        }
    }
}

GDP647_initialize_state_follow = {
    if = {
        limit = { NOT = { has_global_flag = GDP647_state_follow_ready } }
        every_state = { GDP647_refresh_state_follow_scope = yes }
        set_global_flag = GDP647_state_follow_ready
    }
}
