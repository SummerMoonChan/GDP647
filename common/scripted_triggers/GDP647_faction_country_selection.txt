
GDP647_native_faction_filter_active = {
    has_country_flag = GDP647_country_own_faction
    NOT = { has_country_flag = GDP647_manual_selection }
}

GDP647_read_native_faction_page = {
    set_temp_variable = { GDP647_native_faction_slot = 0 }
    if = {
        limit = { is_in_faction = yes }
        if = {
            limit = { is_faction_leader = yes }
            set_temp_variable = { GDP647_native_faction_slot = GDP647_faction_member_slot }
        }
        else = {
            faction_leader = { set_temp_variable = { GDP647_native_faction_slot = GDP647_faction_member_slot } }
        }
    }
    set_temp_variable = { GDP647_faction_native_count = ROOT.GDP647_faction_row_counts^GDP647_native_faction_slot }
    set_temp_variable = { GDP647_faction_native_start = ROOT.GDP647_faction_row_starts^GDP647_native_faction_slot }
    set_temp_variable = { GDP647_faction_native_last = GDP647_faction_native_count }
    subtract_from_temp_variable = { GDP647_faction_native_last = 1 }
    clamp_temp_variable = { var = GDP647_faction_native_last min = 0 }
    set_temp_variable = { GDP647_faction_native_mod = GDP647_faction_native_last }
    modulo_temp_variable = { var = GDP647_faction_native_mod value = 10 }
    subtract_from_temp_variable = { GDP647_faction_native_last = GDP647_faction_native_mod }
    set_temp_variable = { GDP647_faction_native_offset = ROOT.GDP647_page_offsets^0 }
    clamp_temp_variable = { var = GDP647_faction_native_offset min = 0 max = GDP647_faction_native_last }
    always = yes
}

GDP647_append_native_faction_row = {
    if = {
        limit = { check_variable = { GDP647_faction_native_cursor < GDP647_faction_native_end } }
        add_to_temp_array = { array = GDP647_native_faction_rows value = ROOT.GDP647_faction_sorted_rows^GDP647_faction_native_cursor }
        add_to_temp_variable = { GDP647_faction_native_cursor = 1 }
    }
    always = yes
}

GDP647_fill_native_faction_page = {
    clear_temp_array = GDP647_native_faction_rows
    set_temp_variable = { GDP647_faction_native_cursor = GDP647_faction_native_start }
    add_to_temp_variable = { GDP647_faction_native_cursor = GDP647_faction_native_offset }
    set_temp_variable = { GDP647_faction_native_end = GDP647_faction_native_start }
    add_to_temp_variable = { GDP647_faction_native_end = GDP647_faction_native_count }
    set_temp_variable = { GDP647_faction_native_limit = GDP647_faction_native_cursor }
    add_to_temp_variable = { GDP647_faction_native_limit = 10 }
    clamp_temp_variable = { var = GDP647_faction_native_end max = GDP647_faction_native_limit }
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    GDP647_append_native_faction_row = yes
    always = yes
}
